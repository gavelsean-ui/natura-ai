import requests

BASE = "https://www.ebi.ac.uk/europepmc/webservices/rest/search"

def search_europe_pmc(query: str, limit: int = 10):
    if not query: return []
    params = {
        "query": query,
        "format": "json",
        "pageSize": limit,
        "resultType": "core"
    }
    res = requests.get(BASE, params=params, timeout=60)
    res.raise_for_status()
    data = res.json()
    results = data.get("resultList", {}).get("result", [])

    records = []
    for item in results:
        pmid = item.get("pmid", "")
        doi = item.get("doi", "")
        source_url = f"https://pubmed.ncbi.nlm.nih.gov/{pmid}/" if pmid else f"https://europepmc.org/article/{item.get('source', 'MED')}/{item.get('id')}"
        
        # Récupération du type d'article
        pub_types = item.get("pubTypeList", {}).get("pubType", [])
        p_type = pub_types[0] if pub_types else "Journal Article"

        records.append({
            "pmid": pmid,
            "doi": doi,
            "title": item.get("title", "Titre indisponible"),
            "authors": item.get("authorString", "Non précisés"),
            "journal": item.get("journalTitle", "Non précisée"),
            "pub_date": item.get("pubYear", "Non précisée"),
            "abstract": item.get("abstractText", "Aucun résumé disponible."),
            "source": "Europe PMC",
            "type": p_type,
            "source_url": source_url
        })
    return records
